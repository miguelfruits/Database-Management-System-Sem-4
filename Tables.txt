CREATE TABLE dept4
(
deptno number(2,0) primary key,
dname varchar2(13) not null,
loc varchar2(14)
);

CREATE TABLE salgrade4
(
grade number(4) primary key,
losal decimal(10,2),
hiosal decimal(10,2)
);


CREATE TABLE Emp4
(
empno number(4) primary key,
ename varchar2(10) not null,
job varchar2(9),
mgr number(4),
hiredate date Default sysdate,
sal number(7,2) check (sal > 500),
grade number(4) references salgrade4,
comm number(7,2),
deptno number(2,0)references dept4);

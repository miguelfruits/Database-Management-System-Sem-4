SET SERVEROUTPUT ON;

DECLARE
   v_total NUMBER;
BEGIN
   SELECT SUM(marks)
   INTO v_total
   FROM scorecard
   WHERE par_no = 'P0002';

   IF v_total >= 250 THEN
      DBMS_OUTPUT.PUT_LINE('Excellent');
   ELSIF v_total >= 150 THEN
      DBMS_OUTPUT.PUT_LINE('Good');
   ELSE
      DBMS_OUTPUT.PUT_LINE('Needs Improvement');
   END IF;
END;
/

SET SERVEROUTPUT ON;

DECLARE
   v_name participants.part_name%TYPE;
   v_total NUMBER;
BEGIN
   SELECT p.part_name, SUM(s.marks)
   INTO v_name, v_total
   FROM participants p, scorecard s
   WHERE p.part_no = s.par_no
   AND p.part_no = 'P0001'
   GROUP BY p.part_name;

   DBMS_OUTPUT.PUT_LINE('Name: ' || v_name);
   DBMS_OUTPUT.PUT_LINE('Total Marks: ' || v_total);
END;
/


 declare
   p_name participants.part_name%type;
    tot number;
    begin
    select p.part_name,sum(s.marks)
    into p_name,tot
    from participants p, scorecard s
    where p.part_no=s.par_no
    and p.part_no='P0001'
    group by p.part_no;
  dbms_output.put_line('Participants Name  ' || p_name);
  dbms_output.put_line('Marks  ' || tot);
end;

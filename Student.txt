CREATE TABLE SUBJECT_MASTER
(
SUB_CODE VARCHAR2(5) PRIMARY KEY,
SUB_NAME VARCHAR2(20),
CHECK (SUB_CODE LIKE 'S%')
);



CREATE TABLE STUDENT_MASTER
(
ROLL_NO NUMBER(3) PRIMARY KEY,
STUD_NAME VARCHAR2(70),
GENDER VARCHAR2(2),
DOB DATE,
ADDRESS VARCHAR2(150),
CHECK (GENDER IN('F','M'))
);

CREATE TABLE RESULT_MASTER
(
ROLL_NO NUMBER(3) REFERENCES STUDENT_MASTER(ROLL_NO),
SUB_CODE VARCHAR2(5) REFERENCES SUBJECT_MASTER(SUB_CODE),
MARKS NUMBER(3)
);

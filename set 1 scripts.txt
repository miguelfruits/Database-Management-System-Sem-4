drop table Registrations;
drop table Speakers;
drop table Events;

select * from Events;

CREATE TABLE Events (
    EventID NUMBER PRIMARY KEY,
    Name VARCHAR2(25) NOT NULL,
    Description VARCHAR2(45),
    EventDate DATE NOT NULL,  
    Location VARCHAR2(25) NOT NULL
);
CREATE TABLE Speakers (
    SpeakerID NUMBER PRIMARY KEY,
    Name VARCHAR2(25) NOT NULL,
    Bio VARCHAR2(25),
    Expertise VARCHAR2(25) NOT NULL
);
CREATE TABLE Registrations (
    RegistrationID NUMBER PRIMARY KEY,
    EventID NUMBER NOT NULL,
    SpeakerID NUMBER NOT NULL,
    AttendeeName VARCHAR2(25) NOT NULL,
    Email VARCHAR2(25) NOT NULL,
    CONSTRAINT fk_event FOREIGN KEY (EventID) REFERENCES Events(EventID),
    CONSTRAINT fk_speaker FOREIGN KEY (SpeakerID) REFERENCES Speakers(SpeakerID)
);
